// Reviews D123 through actual Runtime acquisition with scripted typed source ports.
// Covers the profile's local ADC/menu path and independent physical-output receipts.
// Synthetic callbacks and configured test windows provide no hardware qualification.
#include "fixtures/app_transaction_fixture.h"
#include "fixtures/app_service_reset/fixture.h"

namespace {
using service_reset_test::Rig;
constexpr auto NONE = service_reset_test::NONE;
constexpr auto START = service_reset_test::START;

std::uint32_t startDrive(Rig& rig) {
    APP_REQUIRE(rig.begin(false, true, false));
    APP_REQUIRE(rig.run(35U)); APP_REQUIRE(rig.select(2U));
    APP_REQUIRE(rig.robot().menu.selection.service == countdown::Service::DRIVE_TEST);
    APP_REQUIRE(rig.run(30U, START)); rig.fake.button_raw = NONE;
    for (unsigned i = 0U; i < 40U; ++i) {
        APP_REQUIRE(rig.next());
        if (rig.robot().lifecycle.gate.start_release) {
            APP_REQUIRE(rig.robot().button_updated && rig.robot().line_available);
            APP_REQUIRE(rig.owner.decisionInput().buttons.explicit_values);
            APP_REQUIRE(rig.owner.decisionInput().line.explicit_values);
            return rig.robot().lifecycle.gate.release_us;
        }
    }
    APP_REQUIRE(false); return 0U;
}
void checkZero(const Rig& rig) {
    CHECK_FALSE(rig.fake.enabled);
    for (auto pulse : rig.fake.pulses) CHECK(pulse == 0U);
}
}

TEST_CASE("B3 B8 B13 D123 reviewer real Runtime ADC release traverses full hold and no-combat drive") {
    Rig rig; const auto released = startDrive(rig);
    rig.fake.opponent_mask = static_cast<std::uint8_t>(127U ^ 0x78U);
    bool go_seen = false; unsigned moving = 0U;
    for (unsigned i = 0U; i < 5250U; ++i) {
        APP_REQUIRE(rig.next()); const auto& result = rig.robot();
        const auto decision = rig.owner.transaction().report().decision_us;
        CHECK_FALSE(result.contact); CHECK_FALSE(result.all_in);
        CHECK(result.contract_faults == 0U);
        if (decision - released < 5100000U) {
            CHECK_FALSE(result.lifecycle.gate.go); CHECK_FALSE(result.outputs.motors_enabled);
            CHECK(rig.fake.highs == 0U); CHECK(rig.fake.nonzero == 0U); checkZero(rig);
        } else {
            go_seen = go_seen || result.lifecycle.gate.go; ++moving;
            CHECK(result.outputs.ui_state == core::State::DRIVE_TEST);
            CHECK(result.opponent_mask == 127U);
            CHECK(result.outputs.motors_enabled);
            CHECK(rig.fake.enabled == (MOTORS_ALLOWED != 0));
            CHECK(rig.owner.transaction().report().applied.feedback.applied_valid);
            if (!MOTORS_ALLOWED) checkZero(rig);
        }
    }
    CHECK(go_seen); CHECK(moving > 100U);
    CHECK(rig.fake.violations == 0U);
}

TEST_CASE("B3 B14 D123 reviewer Runtime source failure inhibits actual active profile outputs") {
    for (bool fail_button : {false, true}) {
        Rig rig; startDrive(rig); APP_REQUIRE(rig.run(5150U));
        APP_REQUIRE(rig.robot().outputs.ui_state == core::State::DRIVE_TEST);
        if (fail_button) rig.fake.buttons_failure = true;
        else rig.fake.opponent_error = 1U;
        rig.next(); checkZero(rig);
        CHECK((rig.robot().outputs.ui_state == core::State::STOPPED ||
              rig.owner.report().phase == app::RuntimePhase::FAULT));
        CHECK_FALSE(rig.robot().lifecycle.gate.go);
    }
}

TEST_CASE("B3 R7 D123 reviewer Runtime empty grants cannot initialize or energize profile") {
    Rig rig; APP_REQUIRE(rig.owner.begin(app::SetupGrants{}));
    for (unsigned i = 0U; i < 1200U; ++i) {
        rig.fake.button_raw = i % 2U ? START : NONE;
        APP_REQUIRE(rig.next()); checkZero(rig);
        CHECK_FALSE(rig.owner.report().initialization_complete);
        CHECK_FALSE(rig.robot().lifecycle.gate.start_release);
        CHECK_FALSE(rig.robot().lifecycle.gate.go);
    }
    CHECK(rig.fake.adc_setups == 0U); CHECK(rig.fake.opp_setups == 0U);
    CHECK(rig.fake.line_setups == 0U); CHECK(rig.fake.highs == 0U);
    CHECK(rig.fake.nonzero == 0U);
}
